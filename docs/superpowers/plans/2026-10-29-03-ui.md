# Plan 3/3 — Interface (PWA React)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal :** Une PWA mobile-first (mode sombre par défaut) qui déchiffre le vault dans le navigateur, affiche tout le patrimoine façon Finary, permet de tout saisir et enregistre par commit via l'API GitHub.

**Architecture :**
- **Stores Zustand** : `app` porte la session, le vault, les cours et la sauvegarde ; `ui` porte le thème et les toasts.
- **Services I/O** : `github`, `pages`, `token`, `live`.
- **Hooks** : `useEngine` et `useHistory` mémoïsent le cœur métier.
- **Formulaires** : un moteur générique (`EntityForm` + `FieldDef[]`) évite d'écrire un formulaire à la main par entité.
- **Graphiques** : Recharts, couleurs lues depuis les variables CSS du thème.

**Tech Stack :** React 18, react-router 6 (HashRouter, compatible GitHub Pages), Zustand 5, Tailwind 3, Recharts 2, vite-plugin-pwa, @zxcvbn-ts.

**Prérequis :** Plans 1 et 2 terminés.

```ts
// file: vite.config.ts
import { createHash } from 'node:crypto';
import { existsSync, readFileSync } from 'node:fs';
import { resolve } from 'node:path';
import { defineConfig, type Plugin } from 'vite';
import react from '@vitejs/plugin-react';
import { VitePWA } from 'vite-plugin-pwa';

/** CSP stricte injectée au build uniquement (le dev de Vite a besoin de scripts inline). */
function csp(): Plugin {
  return {
    name: 'csp', apply: 'build',
    transformIndexHtml(html) {
      const hashes = [...html.matchAll(/<script>([\s\S]*?)<\/script>/g)]
        .map((m) => `'sha256-${createHash('sha256').update(m[1]).digest('base64')}'`).join(' ');
      const policy = [
        "default-src 'self'", `script-src 'self' ${hashes}`, "style-src 'self' 'unsafe-inline'",
        "img-src 'self' data:", "font-src 'self'", "connect-src 'self' https://api.github.com https://api.coingecko.com",
        "worker-src 'self'", "manifest-src 'self'", "base-uri 'self'", "form-action 'none'", "object-src 'none'",
      ].join('; ');
      return html.replace('<meta charset="UTF-8" />', `<meta charset="UTF-8" />\n    <meta http-equiv="Content-Security-Policy" content="${policy}" />`);
    },
  };
}

/** En dev : sert ./data/*.enc sous /data/. */
function devData(): Plugin {
  return {
    name: 'dev-data', apply: 'serve',
    configureServer(server) {
      server.middlewares.use((req, res, next) => {
        const m = req.url?.match(/^\/data\/([\w.-]+\.enc)/);
        if (!m) return next();
        const p = resolve('data', m[1]);
        if (!existsSync(p)) { res.statusCode = 404; return res.end(); }
        res.setHeader('Content-Type', 'application/json');
        res.end(readFileSync(p));
      });
    },
  };
}
```

```ts
// file: scripts/gen-icons.ts
// Génère public/icon-192.png et icon-512.png (sans dépendance) : fond sombre + courbe montante.
import { deflateSync } from 'node:zlib';
import { writeFileSync } from 'node:fs';

function crc32(buf: Buffer): number {
  let c = ~0;
  for (const b of buf) { c ^= b; for (let k = 0; k < 8; k++) c = (c >>> 1) ^ (0xedb88320 & -(c & 1)); }
  return ~c >>> 0;
}
const chunk = (type: string, data: Buffer) => {
  const len = Buffer.alloc(4); len.writeUInt32BE(data.length);
  const td = Buffer.concat([Buffer.from(type), data]);
  const crc = Buffer.alloc(4); crc.writeUInt32BE(crc32(td));
  return Buffer.concat([len, td, crc]);
};

function icon(size: number): Buffer {
  const px = Buffer.alloc(size * size * 4);
  const pts = [[12, 44], [26, 30], [36, 38], [52, 18]].map(([x, y]) => [x * size / 64, y * size / 64]);
  const w = 3 * size / 64;
  const distSeg = (x: number, y: number, [ax, ay]: number[], [bx, by]: number[]) => {
    const t = Math.max(0, Math.min(1, ((x - ax) * (bx - ax) + (y - ay) * (by - ay)) / ((bx - ax) ** 2 + (by - ay) ** 2)));
    return Math.hypot(x - ax - t * (bx - ax), y - ay - t * (by - ay));
  };
  for (let y = 0; y < size; y++) for (let x = 0; x < size; x++) {
    let col = [11, 16, 32];
    for (let k = 0; k < 3; k++) if (distSeg(x, y, pts[k], pts[k + 1]) <= w) col = [129, 140, 248];
    if (Math.hypot(x - pts[3][0], y - pts[3][1]) <= 4 * size / 64) col = [52, 211, 153];
    px.set([...col, 255], (y * size + x) * 4);
  }
  const raw = Buffer.alloc(size * (size * 4 + 1));
  for (let y = 0; y < size; y++) px.copy(raw, y * (size * 4 + 1) + 1, y * size * 4, (y + 1) * size * 4);
  const ihdr = Buffer.alloc(13);
  ihdr.writeUInt32BE(size, 0); ihdr.writeUInt32BE(size, 4); ihdr.set([8, 6, 0, 0, 0], 8);
  return Buffer.concat([Buffer.from([137, 80, 78, 71, 13, 10, 26, 10]), chunk('IHDR', ihdr), chunk('IDAT', deflateSync(raw)), chunk('IEND', Buffer.alloc(0))]);
}
for (const s of [192, 512]) writeFileSync(`public/icon-${s}.png`, icon(s));
console.log('✔ icônes générées');
```

```ts
export const upsert = <T extends { id: string }>(arr: T[], item: T): T[] => {
  const i = arr.findIndex((x) => x.id === item.id);
  if (i < 0) return [...arr, item];
  const c = [...arr]; c[i] = item; return c;
};
export const removeById = <T extends { id: string }>(arr: T[], id: string): T[] => arr.filter((x) => x.id !== id);
export const upsertDated = <T extends { date: string }>(arr: T[] | undefined, item: T): T[] =>
  [...(arr ?? []).filter((x) => x.date !== item.date), item].sort((a, b) => a.date.localeCompare(b.date));
```

```ts
}

export function usePassphraseScore(p: string): number | null {
  const [s, setS] = useState<number | null>(null);
  useEffect(() => {
    let alive = true;
    if (!p) { setS(null); return; }
    load().then((z) => { if (alive) setS(z(p).score); });
    return () => { alive = false; };
  }, [p]);
  return s;
}
```

```ts
// file: src/ui/services/github.ts
export interface RepoRef { owner: string; repo: string; branch: string }

export class GithubError extends Error {
  constructor(public status: number, message: string) { super(message); }
}

const API = 'https://api.github.com';

export const b64encodeUtf8 = (s: string): string => {
  const u8 = new TextEncoder().encode(s);
  let bin = '';
  for (let i = 0; i < u8.length; i += 0x8000) bin += String.fromCharCode(...u8.subarray(i, i + 0x8000));
  return btoa(bin);
};
export const b64decodeUtf8 = (b: string): string =>
  new TextDecoder().decode(Uint8Array.from(atob(b.replace(/\s/g, '')), (c) => c.charCodeAt(0)));

export function repoFromLocation(loc: { hostname: string; pathname: string }): RepoRef | null {
  const m = loc.hostname.match(/^([\w-]+)\.github\.io$/i);
  const repo = loc.pathname.split('/').filter(Boolean)[0];
  return m && repo ? { owner: m[1], repo, branch: 'main' } : null;
}

async function gh(token: string, path: string, init: RequestInit = {}): Promise<Response> {
  const r = await fetch(`${API}${path}`, {
    ...init, cache: 'no-store',
    headers: { Authorization: `Bearer ${token}`, Accept: 'application/vnd.github+json', 'X-GitHub-Api-Version': '2022-11-28', ...(init.headers ?? {}) },
  });
  if (!r.ok) {
    const msg = r.status === 401 || r.status === 403 ? 'Token GitHub invalide ou sans droit d\'écriture'
      : r.status === 404 ? 'Introuvable sur GitHub'
      : r.status === 409 || r.status === 422 ? 'Conflit : le fichier a changé sur GitHub'
      : `Erreur GitHub ${r.status}`;
    throw new GithubError(r.status, msg);
  }
  return r;
}

const contents = (ref: RepoRef, path: string) => `/repos/${ref.owner}/${ref.repo}/contents/${path}`;

export async function getFile(ref: RepoRef, token: string, path: string): Promise<{ sha: string; text: string } | null> {
  try {
    const meta = await (await gh(token, `${contents(ref, path)}?ref=${encodeURIComponent(ref.branch)}`)).json();
    if (meta.encoding === 'base64' && meta.content) return { sha: meta.sha, text: b64decodeUtf8(meta.content) };
    const raw = await gh(token, `/repos/${ref.owner}/${ref.repo}/git/blobs/${meta.sha}`, { headers: { Accept: 'application/vnd.github.raw+json' } });
    return { sha: meta.sha, text: await raw.text() };
  } catch (e) {
    if (e instanceof GithubError && e.status === 404) return null;
    throw e;
  }
}

export async function putFile(ref: RepoRef, token: string, path: string, text: string, sha: string | undefined, message: string): Promise<string> {
  const r = await gh(token, contents(ref, path), {
    method: 'PUT',
    body: JSON.stringify({ message, content: b64encodeUtf8(text), branch: ref.branch, ...(sha ? { sha } : {}) }),
  });
  return (await r.json()).content.sha;
}

/** Commit atomique de plusieurs fichiers (API Git Data). */
export async function commitFiles(ref: RepoRef, token: string, files: { path: string; text: string }[], message: string): Promise<void> {
  const base = `/repos/${ref.owner}/${ref.repo}/git`;
  const head = await (await gh(token, `${base}/ref/heads/${ref.branch}`)).json();
  const commit = await (await gh(token, `${base}/commits/${head.object.sha}`)).json();
  const tree = await (await gh(token, `${base}/trees`, {
    method: 'POST',
    body: JSON.stringify({ base_tree: commit.tree.sha, tree: files.map((f) => ({ path: f.path, mode: '100644', type: 'blob', content: f.text })) }),
  })).json();
  const next = await (await gh(token, `${base}/commits`, { method: 'POST', body: JSON.stringify({ message, tree: tree.sha, parents: [head.object.sha] }) })).json();
  await gh(token, `${base}/refs/heads/${ref.branch}`, { method: 'PATCH', body: JSON.stringify({ sha: next.sha }) });
}

export async function checkAccess(ref: RepoRef, token: string): Promise<boolean> {
  const j = await (await gh(token, `/repos/${ref.owner}/${ref.repo}`)).json();
  return !!j.permissions?.push;
}
```

```ts
// file: src/ui/store/app.ts
import { create } from 'zustand';
import { createSessionKey, decryptJson, encryptJson, unlockSessionKey, DEFAULT_ITER, type Envelope, type SessionKey } from '../../core/crypto';
import { emptyPriceFile, type PriceFile, type Quote } from '../../core/prices';
import { parseVault } from '../../core/schema';
import type { Vault } from '../../core/types';
import { commitFiles, getFile, repoFromLocation, putFile, type RepoRef } from '../services/github';
import { fetchPagesFile } from '../services/pages';
import { clearToken, loadToken, storeToken } from '../services/token';
import { toast } from './ui';

export type Status = 'boot' | 'new' | 'locked' | 'ready';

interface AppState {
  status: Status;
  vaultEnv?: Envelope; pricesEnv?: Envelope;
  sk?: SessionKey; vault?: Vault;
  /** JSON du vault tel que connu sur GitHub (détection de conflit) */
  baseJson?: string;
  prices: PriceFile; live: Record<string, Quote>;
  dirty: number; saving: boolean; token?: string; hide: boolean;
  conflict?: { remote: Vault };

  boot(): Promise<void>;
  unlock(pass: string): Promise<void>;
  createNew(pass: string, initial: Vault, iter?: number): Promise<void>;
  lock(force?: boolean): void;
  mutate(fn: (v: Vault) => void): void;
  replaceVault(v: Vault): void;
  save(): Promise<void>;
  resolveConflict(choice: 'overwrite' | 'reload'): Promise<void>;
  syncRemote(): Promise<void>;
  setToken(token?: string): Promise<void>;
  setHide(h: boolean): void;
  setLive(q: Record<string, Quote>): void;
  changePassphrase(pass: string): Promise<void>;
  repo(): RepoRef | null;
}

async function openPrices(sk: SessionKey, env?: Envelope): Promise<PriceFile> {
  if (!env || env.salt !== sk.salt) return emptyPriceFile();
  try { return { ...emptyPriceFile(), ...(await decryptJson<PriceFile>(sk, env)) }; } catch { return emptyPriceFile(); }
}

const VAULT = 'data/vault.enc';
const PRICES = 'data/prices.enc';

export const useApp = create<AppState>((set, get) => ({
  status: 'boot', prices: emptyPriceFile(), live: {}, dirty: 0, saving: false, hide: false,

  repo() {
    const g = get().vault?.settings.github;
    if (g?.owner && g.repo) return { owner: g.owner, repo: g.repo, branch: g.branch || 'main' };
    return typeof location !== 'undefined' ? repoFromLocation(location) : null;
  },

  async boot() {
    const [v, p] = await Promise.all([fetchPagesFile(VAULT), fetchPagesFile(PRICES)]);
    set({ vaultEnv: v ?? undefined, pricesEnv: p ?? undefined, status: v ? 'locked' : 'new' });
  },

  async unlock(pass) {
    const env = get().vaultEnv;
    if (!env) throw new Error('Aucun patrimoine à déverrouiller');
    const sk = await unlockSessionKey(pass, env);
    const vault = parseVault(await decryptJson(sk, env));
    const prices = await openPrices(sk, get().pricesEnv);
    const token = await loadToken(sk);
    set({ sk, vault, baseJson: JSON.stringify(vault), prices, token, status: 'ready', dirty: 0, hide: vault.settings.hideAmountsByDefault });
    if (token) void get().syncRemote();
  },

  async createNew(pass, initial, iter = DEFAULT_ITER) {
    const sk = await createSessionKey(pass, iter);
    const vault = parseVault(initial);
    set({ sk, vault, baseJson: '', prices: emptyPriceFile(), status: 'ready', dirty: 1, hide: false, token: undefined });
  },

  lock(force = false) {
    if (!force && get().dirty > 0 && !confirm('Des modifications ne sont pas enregistrées. Verrouiller quand même ?')) return;
    set({ sk: undefined, vault: undefined, token: undefined, live: {}, prices: emptyPriceFile(), conflict: undefined, dirty: 0, baseJson: undefined, status: get().vaultEnv ? 'locked' : 'new' });
  },

  mutate(fn) {
    const next = structuredClone(get().vault!);
    fn(next);
    const parsed = parseVault(next);
    set((s) => ({ vault: parsed, dirty: s.dirty + 1 }));
  },

  replaceVault(v) { set((s) => ({ vault: parseVault(v), dirty: s.dirty + 1 })); },

  async save() {
    const { sk, vault, token } = get();
    const ref = get().repo();
    if (!sk || !vault) return;
    if (!token || !ref) { toast('err', 'Configure le dépôt et le token GitHub dans Réglages pour enregistrer.'); return; }
    set({ saving: true });
    try {
      const remote = await getFile(ref, token, VAULT);
      if (remote) {
        const env = JSON.parse(remote.text) as Envelope;
        const remoteVault = parseVault(await decryptJson(sk, env));
        if (JSON.stringify(remoteVault) !== get().baseJson) { set({ conflict: { remote: remoteVault } }); return; }
      }
      const env = await encryptJson(sk, vault);
      await putFile(ref, token, VAULT, JSON.stringify(env) + '\n', remote?.sha, 'chore(data): mise à jour du patrimoine');
      set({ baseJson: JSON.stringify(vault), dirty: 0, vaultEnv: env });
      toast('ok', 'Enregistré ✔ (en ligne d\'ici ~1 min)');
    } catch (e) {
      toast('err', e instanceof Error ? e.message : String(e));
    } finally {
      set({ saving: false });
    }
  },

  async resolveConflict(choice) {
    const c = get().conflict;
    if (!c) return;
    if (choice === 'reload') {
      set({ vault: c.remote, baseJson: JSON.stringify(c.remote), dirty: 0, conflict: undefined });
      toast('info', 'Version GitHub rechargée');
    } else {
      set({ baseJson: JSON.stringify(c.remote), conflict: undefined });
      await get().save();
    }
  },

  async syncRemote() {
    const { sk, token } = get();
    const ref = get().repo();
    if (!sk || !token || !ref) return;
    try {
      const [vf, pf] = await Promise.all([getFile(ref, token, VAULT), getFile(ref, token, PRICES)]);
      if (vf && get().dirty === 0) {
        const env = JSON.parse(vf.text) as Envelope;
        if (env.salt !== sk.salt) { toast('err', 'La passphrase a été changée sur un autre appareil : reverrouille.'); return; }
        const v = parseVault(await decryptJson(sk, env));
        set({ vault: v, baseJson: JSON.stringify(v), vaultEnv: env });
      }
      if (pf) {
        const env = JSON.parse(pf.text) as Envelope;
        set({ prices: await openPrices(sk, env), pricesEnv: env });
      }
    } catch (e) {
      toast('err', `Synchronisation GitHub : ${e instanceof Error ? e.message : e}`);
    }
  },

  async setToken(token) {
    const sk = get().sk;
    if (!sk) return;
    if (token) { await storeToken(sk, token); set({ token }); void get().syncRemote(); }
    else { clearToken(); set({ token: undefined }); }
  },

  setHide(h) { set({ hide: h }); },
  setLive(q) { set({ live: q }); },

  async changePassphrase(pass) {
    const { vault, prices, token } = get();
    const ref = get().repo();
    if (!vault || !token || !ref) throw new Error('Dépôt et token GitHub requis');
    const sk = await createSessionKey(pass);
    const venv = await encryptJson(sk, vault);
    const penv = await encryptJson(sk, prices);
    await commitFiles(ref, token, [
      { path: VAULT, text: JSON.stringify(venv) + '\n' },
      { path: PRICES, text: JSON.stringify(penv) + '\n' },
    ], 'chore(data): changement de passphrase');
    await storeToken(sk, token);
    set({ sk, vaultEnv: venv, pricesEnv: penv, baseJson: JSON.stringify(vault), dirty: 0 });
  },
}));
```

```ts
export function useAutoLock(): void {

    const reset = () => { clearTimeout(t); t = window.setTimeout(lockIfClean, minutes * 60_000); };
    const vis = () => {
      if (document.hidden) hiddenAt = Date.now();
      else if (hiddenAt && Date.now() - hiddenAt > 5 * 60_000) lockIfClean();
    };
    const evs = ['pointerdown', 'keydown', 'scroll', 'touchstart'] as const;
    evs.forEach((ev) => window.addEventListener(ev, reset, { passive: true }));

    reset();
    return () => { clearTimeout(t); evs.forEach((ev) => window.removeEventListener(ev, reset)); document.removeEventListener('visibilitychange', vis); };
  }, [minutes]);
```

```tsx
import { useState, type ReactNode } from 'react';

export type FieldType = 'text' | 'number' | 'percent' | 'date' | 'select' | 'checkbox' | 'multi' | 'textarea' | 'password';
export type Values = Record<string, string | boolean | string[]>;
export interface FieldDef {
  key: string; label: string; type: FieldType;
  options?: { value: string; label: string }[];
  required?: boolean; hint?: string; placeholder?: string;
  show?: (v: Values) => boolean;
}

const parseNum = (s: string): number => Number(s.replace(/[\s  ]/g, '').replace(',', '.'));

export function toValues(fields: FieldDef[], obj: Record<string, unknown>): Values {
  const out: Values = {};
  for (const f of fields) {
    const v = obj[f.key];
    if (f.type === 'checkbox') out[f.key] = !!v;
    else if (f.type === 'multi') out[f.key] = (v as string[]) ?? [];
    else if (f.type === 'percent') out[f.key] = typeof v === 'number' ? String(+(v * 100).toFixed(6)) : '';
    else if (f.type === 'number') out[f.key] = typeof v === 'number' ? String(v) : '';
    else out[f.key] = v == null ? '' : String(v);
  }
  return out;
}

export function fromValues(fields: FieldDef[], values: Values): Record<string, unknown> {
  const out: Record<string, unknown> = {};
  for (const f of fields) {
    const v = values[f.key];
    if (f.show && !f.show(values)) { out[f.key] = undefined; continue; }
    if (f.type === 'checkbox') out[f.key] = !!v;
    else if (f.type === 'multi') out[f.key] = v ?? [];
    else if (f.type === 'number' || f.type === 'percent') {
      const s = String(v ?? '').trim();
      if (!s) { out[f.key] = undefined; continue; }
      const n = parseNum(s);
      if (!isFinite(n)) throw new Error(`Nombre invalide : ${f.label}`);
      out[f.key] = f.type === 'percent' ? n / 100 : n;
    } else out[f.key] = v === '' ? undefined : v;
  }
  return out;
}

export function EntityForm({ fields, initial, onSubmit, onDelete, submitLabel = 'Enregistrer', children }: {
  fields: FieldDef[]; initial: Record<string, unknown>;
  onSubmit: (v: Record<string, unknown>) => void; onDelete?: () => void; submitLabel?: string; children?: ReactNode;
}) {
  const [values, setValues] = useState<Values>(() => toValues(fields, initial));
  const [error, setError] = useState('');
  const set = (k: string, v: Values[string]) => setValues((s) => ({ ...s, [k]: v }));

  function submit(e: React.FormEvent) {
    e.preventDefault();
    setError('');
    try {
      const missing = fields.filter((f) => f.required && (!f.show || f.show(values)) && (values[f.key] === '' || values[f.key] == null));
      if (missing.length) throw new Error(`Champ requis : ${missing.map((f) => f.label).join(', ')}`);
      onSubmit(fromValues(fields, values));
    } catch (err) {
      setError(err instanceof Error ? err.message : String(err));
    }
  }

  return (
    <form onSubmit={submit} className="space-y-3" noValidate>
      {fields.filter((f) => !f.show || f.show(values)).map((f) => {
        const id = `f-${f.key}`;
        const label = <label htmlFor={id} className="mb-1 block text-sm text-mute">{f.label}{f.required && ' *'}</label>;
        const hint = f.hint && <div className="mt-1 text-xs text-mute">{f.hint}</div>;
        if (f.type === 'checkbox') return (
          <label key={f.key} className="flex items-center gap-3 py-1"><input id={id} type="checkbox" checked={!!values[f.key]} onChange={(e) => set(f.key, e.target.checked)} /><span>{f.label}</span></label>
        );
        if (f.type === 'multi') return (
          <fieldset key={f.key}><legend className="mb-1 text-sm text-mute">{f.label}</legend>
            <div className="flex flex-wrap gap-2">{(f.options ?? []).map((o) => {
              const on = (values[f.key] as string[]).includes(o.value);
              return <button type="button" key={o.value} onClick={() => set(f.key, on ? (values[f.key] as string[]).filter((x) => x !== o.value) : [...(values[f.key] as string[]), o.value])}
                className={`rounded-full px-3 py-1 text-sm ${on ? 'bg-accent text-white' : 'bg-surface2 text-mute'}`}>{o.label}</button>;
            })}{!f.options?.length && <span className="text-sm text-mute">—</span>}</div>{hint}</fieldset>
        );
        return (
          <div key={f.key}>
            {label}
            {f.type === 'select' ? (
              <select id={id} value={String(values[f.key])} onChange={(e) => set(f.key, e.target.value)}>
                {!f.required && <option value="">—</option>}
                {f.required && values[f.key] === '' && <option value="" disabled>Choisir…</option>}
                {(f.options ?? []).map((o) => <option key={o.value} value={o.value}>{o.label}</option>)}
              </select>
            ) : f.type === 'textarea' ? (
              <textarea id={id} rows={3} value={String(values[f.key])} onChange={(e) => set(f.key, e.target.value)} placeholder={f.placeholder} />
            ) : (
              <div className="relative">
                <input id={id} value={String(values[f.key])} onChange={(e) => set(f.key, e.target.value)} placeholder={f.placeholder}
                  type={f.type === 'date' ? 'date' : f.type === 'password' ? 'password' : 'text'}
                  inputMode={f.type === 'number' || f.type === 'percent' ? 'decimal' : undefined}
                  autoComplete={f.type === 'password' ? 'off' : undefined} />
                {f.type === 'percent' && <span className="pointer-events-none absolute right-3 top-1/2 -translate-y-1/2 text-mute">%</span>}
              </div>
            )}
            {hint}
          </div>
        );
      })}
      {children}
      {error && <div role="alert" className="rounded-xl bg-down/10 px-3 py-2 text-sm text-down">{error}</div>}
      <div className="flex gap-2 pt-2">
        <Button type="submit" variant="primary" className="flex-1">{submitLabel}</Button>
        {onDelete && <Button type="button" variant="danger" onClick={() => { if (confirm('Supprimer définitivement ?')) onDelete(); }}>Supprimer</Button>}
      </div>

  );
```

```tsx
const useMut = () => useApp((s) => s.mutate);
const useVault = () => useApp((s) => s.vault)!;

export function AccountSheet({ account, onClose }: { account?: Partial<Account>; onClose: Close }) {
  const mutate = useMut();
  const v = useVault();
  const id = account?.id;
  const used = !!id && (v.transactions.some((t) => t.accountId === id) || v.dcaPlans.some((p) => p.accountId === id));
  return (
    <Sheet open onClose={onClose} title={id ? 'Modifier le compte' : 'Nouveau compte'}>

        onDelete={id ? () => { if (used) { toast('err', 'Compte utilisé par des transactions ou un DCA : archive-le plutôt.'); return; } mutate((d) => { d.accounts = removeById(d.accounts, id); }); onClose(); } : undefined} />
    </Sheet>
  );
}
```

```tsx
function PositionSheet({ p, onClose, onAction }: { p: PositionView; onClose: () => void; onAction: (a: 'tx' | 'alert' | 'inst' | 'manual') => void }) {
  const e = useEngine();
  const f = useFmt();
  const prices = useApp((s) => s.prices);
  const inst = e.vault.instruments.find((i) => i.id === p.instrumentId)!;
  const txs = e.transactions.filter((t) => t.accountId === p.accountId && t.instrumentId === p.instrumentId).sort((a, b) => b.date.localeCompare(a.date));
  const firstDate = txs.at(-1)?.date ?? e.today;
  const key = priceKey(inst);
  const series = key ? Object.entries(prices.closes[key] ?? {}).filter(([d]) => d >= firstDate).sort().map(([date, value]) => ({ date, value })) : (inst.manualPrice ?? []).map((m) => ({ date: m.date, value: m.price }));
  const buys = txs.filter((t) => t.type === 'BUY').map((t) => ({ date: series.find((s) => s.date >= t.date)?.date ?? t.date, value: t.unitPrice ?? 0 }));
  const TYPE = { BUY: 'Achat', SELL: 'Vente', DIVIDEND: 'Dividende', DEPOSIT: 'Versement', WITHDRAWAL: 'Retrait', FEE: 'Frais', INTEREST: 'Intérêts' };
  return (
    <Sheet open onClose={onClose} title={p.name} wide>
      <div className="mb-4 grid grid-cols-2 gap-4 sm:grid-cols-4">
        <Stat label="Valeur" value={f.eur(p.value)} />
        <Stat label="+/- value latente" value={<Delta value={p.unrealized}>{f.signed(p.unrealized)}</Delta>} sub={<Delta value={p.unrealizedPct} />} />
        <Stat label="PRU" value={f.eur(p.pru, true)} sub={<span className="text-mute">{qty(p.quantity)} titres</span>} />
        <Stat label="Cours" value={f.money(p.price, p.currency)} sub={<span className="text-mute">{dateFr(p.priceAsOf)}</span>} />
      </div>
      {series.length > 1 && <PriceChart data={series} buys={buys} currency={inst.currency} />}
      <div className="my-4 flex flex-wrap gap-2">
        <Button size="sm" variant="primary" onClick={() => onAction('tx')}><PlusIcon width={16} />Transaction</Button>
        <Button size="sm" onClick={() => onAction('alert')}>Alerte de prix</Button>
        {!key && <Button size="sm" onClick={() => onAction('manual')}>Prix manuel</Button>}
        <Button size="sm" onClick={() => onAction('inst')}>Modifier l'actif</Button>
      </div>
      <SectionTitle>Historique</SectionTitle>

    </Sheet>
  );
}
```

```tsx
import { monthlyEquivalent, projectDca, scheduleDates } from '../../core/dca';
import type { DcaPlan, Transaction } from '../../core/types';

import { Badge, Button, Card, Delta, Empty, PageHeader, Row, SectionTitle, Stat, Tabs } from '../components/ui';
import { EditIcon, PlusIcon } from '../components/icons';
import { useEngine, useFmt } from '../hooks';
import { dateFr, pct, qty } from '../lib/format';
import { DcaPlanSheet, TransactionSheet } from '../sheets/EntitySheets';

const FREQ = { WEEKLY: '/ semaine', BIWEEKLY: '/ 2 semaines', MONTHLY: '/ mois', QUARTERLY: '/ trimestre' };

export default function Dca() {
  const e = useEngine();
  const f = useFmt();
  const [years, setYears] = useState<'5' | '10' | '20'>('10');
  const [sheet, setSheet] = useState<null | { k: 'plan'; id?: string } | { k: 'fix'; tx: Partial<Transaction> }>(null);
  const plans = e.vault.dcaPlans;

  const stats = (p: DcaPlan) => {
    const buys = e.transactions.filter((t) => t.dcaPlanId === p.id && t.type === 'BUY');
    const invested = e.transactions.filter((t) => t.dcaPlanId === p.id && t.type === 'DEPOSIT').reduce((s, t) => s + (t.amount ?? 0), 0);
    const units = buys.reduce((s, t) => s + (t.quantity ?? 0), 0);
    const pos = e.current.positions.find((x) => x.accountId === p.accountId && x.instrumentId === p.instrumentId);
    const value = pos ? units * pos.price * e.book.fxAt(pos.currency, e.today) : 0;
    const next = scheduleDates(p, addDays(e.today, 1), addMonths(e.today, 4))[0];
    return { buys, invested, units, value, next, active: !p.pausedFrom && (!p.endDate || p.endDate >= e.today) };
  };
  const all = plans.map((p) => ({ p, s: stats(p) }));
  const totalInvested = all.reduce((s, x) => s + x.s.invested, 0);
  const totalValue = all.reduce((s, x) => s + x.s.value, 0);
  const monthly = all.filter((x) => x.s.active).reduce((s, x) => s + monthlyEquivalent(x.p), 0);
  const avgReturn = all.length ? all.reduce((s, x) => s + (x.p.expectedAnnualReturn ?? 0.07) * Math.max(1, x.s.value), 0) / all.reduce((s, x) => s + Math.max(1, x.s.value), 0) : 0.07;
  const projection = projectDca(totalValue, monthly, avgReturn, Number(years));

  return (
    <>
      <PageHeader title="DCA" subtitle={`${f.eur(monthly)} investis par mois`} actions={<Button size="sm" variant="primary" onClick={() => setSheet({ k: 'plan' })}><PlusIcon width={16} />Plan</Button>} />
      {!plans.length ? <Empty title="Aucun plan DCA" action={<Button variant="primary" onClick={() => setSheet({ k: 'plan' })}>Créer un plan</Button>}>Déclare tes investissements programmés : les achats sont enregistrés automatiquement au cours de clôture.</Empty> : (
        <>
          <Card className="mb-4">
            <div className="grid grid-cols-2 gap-4 sm:grid-cols-4">
              <Stat label="Total versé" value={f.eur(totalInvested)} />
              <Stat label="Valeur actuelle" value={f.eur(totalValue)} />
              <Stat label="Plus-value" value={<Delta value={totalValue - totalInvested}>{f.signed(totalValue - totalInvested)}</Delta>} sub={<Delta value={totalInvested ? totalValue / totalInvested - 1 : 0} />} />
              <Stat label="Rendement supposé" value={pct(avgReturn, false)} />
            </div>
            <div className="mb-2 mt-5 flex items-center justify-between"><h2 className="font-semibold">Projection</h2>
              <Tabs options={[{ value: '5', label: '5 ans' }, { value: '10', label: '10 ans' }, { value: '20', label: '20 ans' }]} value={years} onChange={setYears} /></div>
            <ProjectionChart data={projection} hide={f.hide} />
            <div className="text-sm text-mute">Dans {years} ans : <b className="text-ink">{f.eur(projection.at(-1)!.value)}</b> dont {f.eur(projection.at(-1)!.invested)} versés (hypothèse, non garantie).</div>
          </Card>

          {e.pendingDca.length > 0 && (
            <>
              <SectionTitle>Échéances sans cours</SectionTitle>
              <Card>{e.pendingDca.map((x) => { const p = plans.find((pl) => pl.id === x.planId)!; return (
                <Row key={x.planId + x.date} left={p.name} sub={dateFr(x.date)} right={<Button size="sm" onClick={() => setSheet({ k: 'fix', tx: { accountId: p.accountId, instrumentId: p.instrumentId, type: 'BUY', date: x.date, dcaPlanId: p.id, dcaDate: x.date } })}>Saisir</Button>} />); })}</Card>
            </>
          )}

          <SectionTitle>Plans</SectionTitle>
          <div className="grid gap-4 md:grid-cols-2">
            {all.map(({ p, s }) => (
              <Card key={p.id}>
                <div className="flex items-start justify-between">
                  <div><div className="font-semibold">{p.name} {!s.active && <Badge>{p.pausedFrom ? 'en pause' : 'terminé'}</Badge>}</div>
                    <div className="text-xs text-mute">{f.eur(p.amount)} {FREQ[p.frequency]} · {e.vault.accounts.find((a) => a.id === p.accountId)?.name}</div></div>
                  <Button size="sm" variant="ghost" onClick={() => setSheet({ k: 'plan', id: p.id })} aria-label="Modifier"><EditIcon /></Button>
                </div>
                <div className="mt-3 grid grid-cols-3 gap-3">
                  <Stat label="Versé" value={f.eur(s.invested)} />
                  <Stat label="Valeur" value={f.eur(s.value)} sub={<Delta value={s.invested ? s.value / s.invested - 1 : 0} />} />
                  <Stat label="Titres" value={qty(s.units)} />
                </div>
                <div className="mt-2 text-xs text-mute">{s.next ? `Prochaine échéance : ${dateFr(s.next)}` : 'Pas d\'échéance à venir'} · {s.buys.length} achats</div>
                <details className="mt-2">
                  <summary className="cursor-pointer text-sm text-accent">Derniers achats</summary>
                  {s.buys.slice(-6).reverse().map((t) => (
                    <Row key={t.id} left={dateFr(t.date)} sub={t.virtual ? 'auto (clôture)' : 'corrigé'} right={`${qty(t.quantity ?? 0)} × ${f.money(t.unitPrice ?? 0, e.vault.instruments.find((i) => i.id === p.instrumentId)?.currency ?? 'EUR')}`}
                      subRight={<button className="text-accent" onClick={() => setSheet({ k: 'fix', tx: t.virtual ? { ...t, id: undefined, virtual: undefined } : t })}>Corriger</button>} />
                  ))}
                </details>
              </Card>
            ))}
          </div>
        </>
      )}
      {sheet?.k === 'plan' && <DcaPlanSheet plan={plans.find((p) => p.id === sheet.id)} onClose={() => setSheet(null)} />}
      {sheet?.k === 'fix' && <TransactionSheet tx={sheet.tx} onClose={() => setSheet(null)} />}
    </>
  );
}
```

```tsx
import { useMemo, useState } from 'react';
import type { Transaction } from '../../core/types';
import { Badge, Button, Card, Empty, PageHeader, Row } from '../components/ui';
import { PlusIcon } from '../components/icons';
import { useEngine, useFmt } from '../hooks';
import { dateFr, monthFr, qty } from '../lib/format';
import { TransactionSheet } from '../sheets/EntitySheets';

const TYPE: Record<Transaction['type'], string> = { BUY: 'Achat', SELL: 'Vente', DIVIDEND: 'Dividende', DEPOSIT: 'Versement', WITHDRAWAL: 'Retrait', FEE: 'Frais', INTEREST: 'Intérêts' };

export default function Transactions() {
  const e = useEngine();
  const f = useFmt();
  const [acc, setAcc] = useState('');
  const [type, setType] = useState('');
  const [q, setQ] = useState('');
  const [auto, setAuto] = useState(true);
  const [limit, setLimit] = useState(100);
  const [sheet, setSheet] = useState<null | Partial<Transaction>>(null);
  const inst = (id?: string) => e.vault.instruments.find((i) => i.id === id);
  const accName = (id: string) => e.vault.accounts.find((a) => a.id === id)?.name ?? '?';

  const list = useMemo(() => e.transactions
    .filter((t) => (!acc || t.accountId === acc) && (!type || t.type === type) && (auto || !t.virtual))
    .filter((t) => !q || `${inst(t.instrumentId)?.name ?? ''} ${t.note ?? ''} ${accName(t.accountId)}`.toLowerCase().includes(q.toLowerCase()))
    .sort((a, b) => b.date.localeCompare(a.date)), [e, acc, type, q, auto]);

  const months = new Map<string, Transaction[]>();
  for (const t of list.slice(0, limit)) months.set(t.date.slice(0, 7), [...(months.get(t.date.slice(0, 7)) ?? []), t]);

  return (
    <>
      <PageHeader title="Transactions" subtitle={`${list.length} opérations`} actions={<Button size="sm" variant="primary" onClick={() => setSheet({})}><PlusIcon width={16} />Transaction</Button>} />
      <div className="mb-4 grid grid-cols-2 gap-2 md:grid-cols-4">
        <select value={acc} onChange={(x) => setAcc(x.target.value)} aria-label="Compte"><option value="">Tous les comptes</option>{e.vault.accounts.map((a) => <option key={a.id} value={a.id}>{a.name}</option>)}</select>
        <select value={type} onChange={(x) => setType(x.target.value)} aria-label="Type"><option value="">Tous les types</option>{Object.entries(TYPE).map(([k, l]) => <option key={k} value={k}>{l}</option>)}</select>
        <input value={q} onChange={(x) => setQ(x.target.value)} placeholder="Rechercher…" aria-label="Rechercher" />
        <label className="flex items-center gap-2 text-sm"><input type="checkbox" checked={auto} onChange={(x) => setAuto(x.target.checked)} />DCA auto</label>
      </div>
      {!list.length ? <Empty title="Aucune transaction" /> : [...months].map(([m, txs]) => (
        <div key={m} className="mb-4">
          <div className="mb-1 text-xs font-semibold uppercase text-mute">{monthFr(`${m}-01`)}</div>
          <Card className="py-1">{txs.map((t) => {
            const i = inst(t.instrumentId);
            const sign = ['BUY', 'WITHDRAWAL', 'FEE'].includes(t.type) ? -1 : 1;
            const amt = t.quantity ? t.quantity * (t.unitPrice ?? 0) * (t.fxRate ?? 1) : t.amount ?? 0;
            return (
              <Row key={t.id} onClick={() => setSheet(t.virtual ? { ...t, id: undefined, virtual: undefined } : t)}
                left={<>{TYPE[t.type]}{i ? ` · ${i.name}` : ''} {t.virtual && <Badge tone="accent">DCA auto</Badge>}</>}
                sub={`${dateFr(t.date)} · ${accName(t.accountId)}${t.quantity ? ` · ${qty(t.quantity)} × ${f.money(t.unitPrice ?? 0, i?.currency ?? 'EUR')}` : ''}`}
                right={<span className={sign > 0 ? 'text-up' : ''}>{sign > 0 ? '+' : '−'}{f.eur(Math.abs(amt), true)}</span>} />
            );
          })}</Card>
        </div>
      ))}
      {list.length > limit && <Button className="w-full" onClick={() => setLimit(limit + 200)}>Afficher plus</Button>}
      {sheet && <TransactionSheet tx={sheet} onClose={() => setSheet(null)} />}
    </>
  );
}
```
