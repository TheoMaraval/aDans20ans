# Plan 1/3 — Fondations et cœur métier

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal :** Construire `src/core`, une bibliothèque TypeScript pure et testée. Elle déchiffre les données, valorise tout le patrimoine à n'importe quelle date, génère les DCA, calcule les performances, l'historique et les alertes, et prépare les données du rapport hebdo.

**Architecture :** Des modules sans dépendance au DOM, utilisables tels quels par le navigateur (React) et par Node 22 (scripts des GitHub Actions). Le schéma zod sert de source unique pour les types. `engine.ts` est la façade qui assemble le tout.

**Tech Stack :** TypeScript 5.6 strict, zod 3, Vitest 2, WebCrypto (`globalThis.crypto.subtle`).

**Spec :** `docs/superpowers/specs/2026-10-29-dashboard-patrimoine-design.md` (sections 4, 5, 7, 8).

**Conventions :**
- Dates `YYYY-MM-DD` en UTC, gérées par `dates.ts`.
- Montants en `number`, arrondis seulement à l'affichage.
- Toutes les sorties de valorisation sont en EUR.
- `fees` et `amount` d'une transaction sont dans la devise du **compte** ; `unitPrice` est dans la devise de l'**instrument**.
- Chaque bloc de code commence par `// file: <chemin>`.

// file: src/core/dates.ts
export type ISODate = string;
const DAY = 86_400_000;

export const toDate = (d: ISODate): Date => new Date(`${d.slice(0, 10)}T00:00:00Z`);
export const fmt = (d: Date): ISODate => d.toISOString().slice(0, 10);
export const addDays = (d: ISODate, n: number): ISODate => fmt(new Date(toDate(d).getTime() + n * DAY));
export const daysBetween = (a: ISODate, b: ISODate): number =>
  Math.round((toDate(b).getTime() - toDate(a).getTime()) / DAY);
export const yearsBetween = (a: ISODate, b: ISODate): number => daysBetween(a, b) / 365.25;

export function addMonths(d: ISODate, n: number): ISODate {
  const x = toDate(d);
  const day = x.getUTCDate();
  const t = new Date(Date.UTC(x.getUTCFullYear(), x.getUTCMonth() + n, 1));
  const last = new Date(Date.UTC(t.getUTCFullYear(), t.getUTCMonth() + 1, 0)).getUTCDate();
  t.setUTCDate(Math.min(day, last));
  return fmt(t);
}

export function eachDay(from: ISODate, to: ISODate): ISODate[] {
  const out: ISODate[] = [];
  for (let d = from; d <= to; d = addDays(d, 1)) out.push(d);
  return out;
}

export const isWeekend = (d: ISODate): boolean => {
  const w = toDate(d).getUTCDay();
  return w === 0 || w === 6;
};
/** 1 = lundi … 7 = dimanche */
export const isoWeekday = (d: ISODate): number => ((toDate(d).getUTCDay() + 6) % 7) + 1;

/** Date du jour à Paris. */
export const todayParis = (now: Date = new Date()): ISODate =>
  now.toLocaleDateString('sv-SE', { timeZone: 'Europe/Paris' });

export type Series = Record<ISODate, number>;
// Les séries sont considérées immuables une fois construites (le cache de clés en dépend).
const keyCache = new WeakMap<Series, ISODate[]>();
function sortedKeys(s: Series): ISODate[] {
  let k = keyCache.get(s);
  if (!k) { k = Object.keys(s).sort(); keyCache.set(s, k); }
  return k;
}

export function lastOnOrBefore(s: Series | undefined, d: ISODate): { date: ISODate; value: number } | null {
  if (!s) return null;
  const k = sortedKeys(s);
  let lo = 0, hi = k.length - 1, ans = -1;
  while (lo <= hi) {
    const mid = (lo + hi) >> 1;
    if (k[mid] <= d) { ans = mid; lo = mid + 1; } else hi = mid - 1;
  }
  return ans < 0 ? null : { date: k[ans], value: s[k[ans]] };
}

export function firstOnOrAfter(s: Series | undefined, d: ISODate): { date: ISODate; value: number } | null {
  if (!s) return null;
  const k = sortedKeys(s);
  let lo = 0, hi = k.length - 1, ans = -1;
  while (lo <= hi) {
    const mid = (lo + hi) >> 1;
    if (k[mid] >= d) { ans = mid; hi = mid - 1; } else lo = mid + 1;
  }
  return ans < 0 ? null : { date: k[ans], value: s[k[ans]] };
}

export function lastEntryOnOrBefore<T extends { date: ISODate }>(arr: T[] | undefined, d: ISODate): T | undefined {
  let best: T | undefined;
  for (const e of arr ?? []) if (e.date <= d && (!best || e.date >= best.date)) best = e;
  return best;
}

export const vaultSchema = z.object({
  version: z.literal(1),
  settings: settingsSchema.default({}),
  accounts: z.array(accountSchema).default([]),
  instruments: z.array(instrumentSchema).default([]),
  transactions: z.array(transactionSchema).default([]),
  dcaPlans: z.array(dcaPlanSchema).default([]),
  realEstate: z.array(realEstateSchema).default([]),
  loans: z.array(loanSchema).default([]),
  physicalAssets: z.array(physicalAssetSchema).default([]),
  priceAlerts: z.array(priceAlertSchema).default([]),
}).superRefine((v, ctx) => {
  const accounts = new Set(v.accounts.map((a) => a.id));
  const instruments = new Set(v.instruments.map((i) => i.id));
  const loans = new Set(v.loans.map((l) => l.id));
  const plans = new Set(v.dcaPlans.map((p) => p.id));
  const err = (path: (string | number)[], message: string) => ctx.addIssue({ code: z.ZodIssueCode.custom, path, message });
  v.transactions.forEach((t, i) => {
    if (!accounts.has(t.accountId)) err(['transactions', i, 'accountId'], `compte inconnu ${t.accountId}`);
    if (['BUY', 'SELL', 'DIVIDEND'].includes(t.type) && !t.instrumentId) err(['transactions', i], `${t.type} sans instrument`);
    if (t.instrumentId && !instruments.has(t.instrumentId)) err(['transactions', i, 'instrumentId'], `instrument inconnu ${t.instrumentId}`);
    if ((t.type === 'BUY' || t.type === 'SELL') && (t.quantity === undefined || t.unitPrice === undefined)) err(['transactions', i], `${t.type} sans quantité ou prix`);
    if (['DIVIDEND', 'DEPOSIT', 'WITHDRAWAL', 'FEE', 'INTEREST'].includes(t.type) && t.amount === undefined) err(['transactions', i], `${t.type} sans montant`);
    if (t.dcaPlanId && !plans.has(t.dcaPlanId)) err(['transactions', i, 'dcaPlanId'], `plan DCA inconnu ${t.dcaPlanId}`);
  });
  v.dcaPlans.forEach((p, i) => {
    if (!accounts.has(p.accountId)) err(['dcaPlans', i, 'accountId'], `compte inconnu ${p.accountId}`);
    if (!instruments.has(p.instrumentId)) err(['dcaPlans', i, 'instrumentId'], `instrument inconnu ${p.instrumentId}`);
  });
  v.realEstate.forEach((r, i) => r.loanIds.forEach((l) => { if (!loans.has(l)) err(['realEstate', i, 'loanIds'], `prêt inconnu ${l}`); }));
  v.physicalAssets.forEach((r, i) => r.loanIds.forEach((l) => { if (!loans.has(l)) err(['physicalAssets', i, 'loanIds'], `prêt inconnu ${l}`); }));
  v.priceAlerts.forEach((a, i) => { if (!instruments.has(a.instrumentId)) err(['priceAlerts', i], `instrument inconnu ${a.instrumentId}`); });
  v.settings.emergencyFundAccountIds.forEach((a) => { if (!accounts.has(a)) err(['settings', 'emergencyFundAccountIds'], `compte inconnu ${a}`); });
});

export class VaultValidationError extends Error {
  constructor(public issues: z.ZodIssue[]) {
    super(issues.map((i) => `${i.path.join('.')}: ${i.message}`).join('\n'));
  }
}

export function parseVault(input: unknown) {
  const r = vaultSchema.safeParse(input);
  if (!r.success) throw new VaultValidationError(r.error.issues);
  return r.data;
}

// file: src/core/crypto.ts
export const DEFAULT_ITER = 600_000;

export interface Envelope { v: 1; kdf: 'PBKDF2-SHA256'; iter: number; salt: string; iv: string; ct: string }
export interface SessionKey { key: CryptoKey; salt: string; iter: number }

export class DecryptError extends Error {
  constructor() { super('Passphrase incorrecte ou fichier corrompu'); }
}

const subtle = () => globalThis.crypto.subtle;
const enc = new TextEncoder();
const dec = new TextDecoder();

export function toB64(u8: Uint8Array): string {
  let s = '';
  for (let i = 0; i < u8.length; i += 0x8000) s += String.fromCharCode(...u8.subarray(i, i + 0x8000));
  return btoa(s);
}
export const fromB64 = (s: string): Uint8Array => Uint8Array.from(atob(s), (c) => c.charCodeAt(0));
export const randomBytes = (n: number): Uint8Array => globalThis.crypto.getRandomValues(new Uint8Array(n));

export function isEnvelope(x: unknown): x is Envelope {
  const e = x as Envelope;
  return !!e && e.v === 1 && e.kdf === 'PBKDF2-SHA256' && typeof e.salt === 'string' && typeof e.iv === 'string' && typeof e.ct === 'string';
}

async function deriveKey(passphrase: string, salt: Uint8Array, iter: number): Promise<CryptoKey> {
  const base = await subtle().importKey('raw', enc.encode(passphrase.normalize('NFC')), 'PBKDF2', false, ['deriveKey']);
  return subtle().deriveKey(
    { name: 'PBKDF2', hash: 'SHA-256', salt, iterations: iter },
    base, { name: 'AES-GCM', length: 256 }, false, ['encrypt', 'decrypt'],
  );
}

export async function createSessionKey(passphrase: string, iter = DEFAULT_ITER): Promise<SessionKey> {
  const salt = randomBytes(16);
  return { key: await deriveKey(passphrase, salt, iter), salt: toB64(salt), iter };
}

export async function unlockSessionKey(passphrase: string, env: Envelope): Promise<SessionKey> {
  return { key: await deriveKey(passphrase, fromB64(env.salt), env.iter), salt: env.salt, iter: env.iter };
}

export async function encryptJson(sk: SessionKey, data: unknown): Promise<Envelope> {
  const iv = randomBytes(12);
  const ct = await subtle().encrypt({ name: 'AES-GCM', iv }, sk.key, enc.encode(JSON.stringify(data)));
  return { v: 1, kdf: 'PBKDF2-SHA256', iter: sk.iter, salt: sk.salt, iv: toB64(iv), ct: toB64(new Uint8Array(ct)) };
}

export async function decryptJson<T>(sk: SessionKey, env: Envelope): Promise<T> {
  if (env.salt !== sk.salt) throw new Error('Enveloppe chiffrée avec un autre sel (passphrase changée ?)');
  try {
    const pt = await subtle().decrypt({ name: 'AES-GCM', iv: fromB64(env.iv) }, sk.key, fromB64(env.ct));
    return JSON.parse(dec.decode(pt)) as T;
  } catch {
    throw new DecryptError();
  }
}

// file: src/core/prices.ts
import { addDays, firstOnOrAfter, lastEntryOnOrBefore, lastOnOrBefore, type ISODate, type Series } from './dates';
import type { Instrument, PriceSource, Transaction } from './types';

export interface Quote { price: number; currency: string; asOf: string; changePct1d?: number }

export interface PriceFile {
  updatedAt: string;
  quotes: Record<string, Quote>;
  /** EUR par unité de devise, dernier connu */
  fx: Record<string, number>;
  closes: Record<string, Series>;
  fxCloses: Record<string, Series>;
  errors: { symbol: string; message: string; at: string }[];
}

export const emptyPriceFile = (): PriceFile => ({ updatedAt: '', quotes: {}, fx: {}, closes: {}, fxCloses: {}, errors: [] });

export const priceKey = (i: Pick<Instrument, 'yahooSymbol' | 'coingeckoId'>): string | null =>
  i.yahooSymbol ?? (i.coingeckoId ? `cg:${i.coingeckoId}` : null);

export interface PricePoint { price: number; asOf: string; source: PriceSource }

export interface PriceBook {
  today: ISODate;
  /** Cours en devise de l'instrument, dernier connu à la date. */
  priceAt(i: Instrument, date: ISODate): PricePoint | null;
  /** Clôture du jour ou du premier jour de cotation suivant (≤ maxDays). */
  closeOnOrAfter(i: Instrument, date: ISODate, maxDays?: number): PricePoint | null;
  /** EUR par unité de devise. */
  fxAt(currency: string, date: ISODate): number;
  changePct1d(i: Instrument): number | undefined;
}

export function createPriceBook(pf: PriceFile, transactions: Transaction[], today: ISODate, live: Record<string, Quote> = {}): PriceBook {
  const txPrices = new Map<string, Series>();
  for (const t of transactions) {
    if ((t.type === 'BUY' || t.type === 'SELL') && t.instrumentId && t.unitPrice !== undefined) {
      const s = txPrices.get(t.instrumentId) ?? {};
      s[t.date] = t.unitPrice;
      txPrices.set(t.instrumentId, s);
    }
  }
  const quoteFor = (i: Instrument): { q: Quote; source: PriceSource } | null => {
    const k = priceKey(i);
    if (!k) return null;
    if (live[k]) return { q: live[k], source: 'live' };
    if (pf.quotes[k]) return { q: pf.quotes[k], source: 'quote' };
    return null;
  };
  const fallback = (i: Instrument, date: ISODate): PricePoint | null => {
    const m = lastEntryOnOrBefore(i.manualPrice, date);
    if (m) return { price: m.price, asOf: m.date, source: 'manual' };
    const t = lastOnOrBefore(txPrices.get(i.id), date);
    return t ? { price: t.value, asOf: t.date, source: 'transaction' } : null;
  };

  return {
    today,
    priceAt(i, date) {
      if (date >= today) {
        const q = quoteFor(i);
        if (q) return { price: q.q.price, asOf: q.q.asOf, source: q.source };
      }
      const k = priceKey(i);
      const c = k ? lastOnOrBefore(pf.closes[k], date) : null;
      if (c) return { price: c.value, asOf: c.date, source: 'close' };
      return fallback(i, date);
    },
    closeOnOrAfter(i, date, maxDays = 7) {
      const k = priceKey(i);
      if (!k) return fallback(i, date);
      const limit = addDays(date, maxDays);
      const c = firstOnOrAfter(pf.closes[k], date);
      if (c && c.date <= limit) return { price: c.value, asOf: c.date, source: 'close' };
      const q = quoteFor(i);
      const qd = q?.q.asOf.slice(0, 10);
      if (q && qd && qd >= date && qd <= limit) return { price: q.q.price, asOf: q.q.asOf, source: q.source };
      return null;
    },
    fxAt(currency, date) {
      if (currency === 'EUR') return 1;
      if (date >= today && pf.fx[currency]) return pf.fx[currency];
      return lastOnOrBefore(pf.fxCloses[currency], date)?.value
        ?? firstOnOrAfter(pf.fxCloses[currency], date)?.value
        ?? pf.fx[currency] ?? 1;
    },
    changePct1d(i) { return quoteFor(i)?.q.changePct1d; },
  };
}

// file: src/core/positions.ts
import type { ISODate } from './dates';
import type { PriceBook } from './prices';
import type { Account, Id, Instrument, Transaction } from './types';

export interface PositionState {
  accountId: Id; instrumentId: Id;
  quantity: number;
  /** EUR, frais d'achat inclus */
  costBasis: number;
  realized: number; dividends: number;
  firstDate: ISODate;
}

export interface LedgerResult {
  positions: PositionState[];
  /** cash EUR par compte, issu des transactions */
  cash: Record<Id, number>;
  /** versements − retraits EUR par compte */
  netDeposits: Record<Id, number>;
  errors: string[];
}

const EPS = 1e-9;

export function runLedger(
  txs: Transaction[], accounts: Map<Id, Account>, instruments: Map<Id, Instrument>, book: PriceBook, until?: ISODate,
): LedgerResult {
  const sorted = txs.filter((t) => !until || t.date <= until).sort((a, b) => (a.date < b.date ? -1 : a.date > b.date ? 1 : 0));
  const pos = new Map<string, PositionState>();
  const cash: Record<Id, number> = {};
  const netDeposits: Record<Id, number> = {};
  const errors: string[] = [];

  for (const t of sorted) {
    const accFx = book.fxAt(accounts.get(t.accountId)?.currency ?? 'EUR', t.date);
    const amount = (t.amount ?? 0) * accFx;
    const fees = (t.fees ?? 0) * accFx;
    cash[t.accountId] ??= 0;
    netDeposits[t.accountId] ??= 0;

    if (t.type === 'BUY' || t.type === 'SELL' || t.type === 'DIVIDEND') {
      const i = instruments.get(t.instrumentId!);
      if (!i) { errors.push(`${t.id}: instrument inconnu`); continue; }
      const key = `${t.accountId}|${i.id}`;
      const p = pos.get(key) ?? { accountId: t.accountId, instrumentId: i.id, quantity: 0, costBasis: 0, realized: 0, dividends: 0, firstDate: t.date };
      pos.set(key, p);
      const fx = t.fxRate ?? book.fxAt(i.currency, t.date);
      const gross = (t.quantity ?? 0) * (t.unitPrice ?? 0) * fx;

      if (t.type === 'BUY') {
        p.quantity += t.quantity!;
        p.costBasis += gross + fees;
        cash[t.accountId] -= gross + fees;
      } else if (t.type === 'SELL') {
        let q = t.quantity!;
        if (q > p.quantity + EPS) {
          errors.push(`${t.id}: vente de ${q} > ${p.quantity} détenus (${i.name})`);
          q = p.quantity;
        }
        const pru = p.quantity > EPS ? p.costBasis / p.quantity : 0;
        const proceeds = q * (t.unitPrice ?? 0) * fx;
        p.realized += proceeds - fees - q * pru;
        p.costBasis -= q * pru;
        p.quantity -= q;
        if (p.quantity < EPS) { p.quantity = 0; p.costBasis = 0; }
        cash[t.accountId] += proceeds - fees;
      } else {
        p.dividends += amount;
        cash[t.accountId] += amount - fees;
      }
      continue;
    }
    switch (t.type) {
      case 'DEPOSIT': cash[t.accountId] += amount; netDeposits[t.accountId] += amount; break;
      case 'WITHDRAWAL': cash[t.accountId] -= amount; netDeposits[t.accountId] -= amount; break;
      case 'FEE': cash[t.accountId] -= amount + fees; break;
      case 'INTEREST': cash[t.accountId] += amount - fees; break;
    }
  }
  return { positions: [...pos.values()], cash, netDeposits, errors };
}

import { addDays, addMonths, isoWeekday, type ISODate } from './dates';
import type { PriceBook } from './prices';
import type { DcaPlan, Id, PendingDca, Transaction, Vault } from './types';

/** Échéances du plan dans [from, to] (from par défaut = startDate). */
export function scheduleDates(plan: DcaPlan, from: ISODate | undefined, to: ISODate): ISODate[] {
  const lower = from && from > plan.startDate ? from : plan.startDate;
  let upper = to;
  if (plan.endDate && plan.endDate < upper) upper = plan.endDate;
  if (plan.pausedFrom) { const p = addDays(plan.pausedFrom, -1); if (p < upper) upper = p; }
  const out: ISODate[] = [];
  if (plan.frequency === 'MONTHLY' || plan.frequency === 'QUARTERLY') {
    const step = plan.frequency === 'MONTHLY' ? 1 : 3;
    const dom = String(plan.dayOfMonth ?? 1).padStart(2, '0');
    let d = `${plan.startDate.slice(0, 7)}-${dom}`;
    if (d < plan.startDate) d = addMonths(d, 1);
    for (; d <= upper; d = addMonths(d, step)) if (d >= lower) out.push(d);
  } else {
    const step = plan.frequency === 'WEEKLY' ? 7 : 14;
    let d = plan.startDate;
    while (isoWeekday(d) !== (plan.dayOfWeek ?? 1)) d = addDays(d, 1);
    for (; d <= upper; d = addDays(d, step)) if (d >= lower) out.push(d);
  }
  return out;
}

export interface ExpandResult { transactions: Transaction[]; pending: PendingDca[] }

/** Transactions réelles + achats DCA virtuels + versements implicites. */
export function expandTransactions(vault: Vault, book: PriceBook, today: ISODate): ExpandResult {
  const out: Transaction[] = [...vault.transactions];
  const pending: PendingDca[] = [];
  const instruments = new Map(vault.instruments.map((i) => [i.id, i]));
  const corrections = new Map<string, Transaction>();
  for (const t of vault.transactions) if (t.dcaPlanId && t.dcaDate) corrections.set(`${t.dcaPlanId}:${t.dcaDate}`, t);

  for (const plan of vault.dcaPlans) {
    const i = instruments.get(plan.instrumentId);
    if (!i) continue;
    for (const d of scheduleDates(plan, undefined, today)) {
      const corr = corrections.get(`${plan.id}:${d}`);
      if (corr) {
        const fx = corr.fxRate ?? book.fxAt(i.currency, corr.date);
        out.push(deposit(plan.id, d, plan.accountId, corr.date, (corr.quantity ?? 0) * (corr.unitPrice ?? 0) * fx + (corr.fees ?? 0)));
        continue;
      }
      const pp = book.closeOnOrAfter(i, d);
      if (!pp) { pending.push({ planId: plan.id, date: d }); continue; }
      const date = pp.asOf.slice(0, 10);
      const fx = book.fxAt(i.currency, date);
      const fees = plan.amount * (plan.feesPct ?? 0);
      const raw = (plan.amount - fees) / (pp.price * fx);
      const quantity = plan.fractional ? raw : Math.floor(raw);
      out.push(deposit(plan.id, d, plan.accountId, date, plan.amount));
      if (quantity > 0) {
        out.push({
          id: `dca:${plan.id}:${d}`, accountId: plan.accountId, date, type: 'BUY', instrumentId: i.id,
          quantity, unitPrice: pp.price, fees, fxRate: fx, dcaPlanId: plan.id, dcaDate: d, virtual: true,
        });
      }
    }
  }
  return { transactions: out, pending };
}

function deposit(planId: Id, d: ISODate, accountId: Id, date: ISODate, amount: number): Transaction {
  return { id: `dcadep:${planId}:${d}`, accountId, date, type: 'DEPOSIT', amount, dcaPlanId: planId, dcaDate: d, virtual: true };
}

const PER_MONTH: Record<DcaPlan['frequency'], number> = { WEEKLY: 52 / 12, BIWEEKLY: 26 / 12, MONTHLY: 1, QUARTERLY: 1 / 3 };
export const monthlyEquivalent = (plan: DcaPlan): number => plan.amount * PER_MONTH[plan.frequency];

export interface ProjectionPoint { year: number; invested: number; value: number }

/** Projection : versements mensuels capitalisés, un point par année (année 0 = aujourd'hui). */
export function projectDca(startValue: number, monthly: number, annualReturn: number, years: number): ProjectionPoint[] {
  const r = Math.pow(1 + annualReturn, 1 / 12) - 1;
  let value = startValue, invested = startValue;
  const out: ProjectionPoint[] = [{ year: 0, invested, value }];
  for (let m = 1; m <= years * 12; m++) {
    value = value * (1 + r) + monthly;
    invested += monthly;
    if (m % 12 === 0) out.push({ year: m / 12, invested: Math.round(invested * 100) / 100, value: Math.round(value * 100) / 100 });
  }
  return out;
}

// file: src/core/loans.ts
import { addMonths, lastEntryOnOrBefore, type ISODate } from './dates';
import type { Loan } from './types';

export function monthlyPayment(principal: number, annualRate: number, n: number): number {
  const i = annualRate / 12;
  return i === 0 ? principal / n : (principal * i) / (1 - Math.pow(1 + i, -n));
}

/** Solde après k mensualités M depuis un capital B. */
function balanceAfter(B: number, i: number, M: number, k: number): number {
  const b = i === 0 ? B - M * k : B * Math.pow(1 + i, k) - (M * (Math.pow(1 + i, k) - 1)) / i;
  return Math.max(0, b);
}

/** Nombre d'échéances (dates startDate + k mois) dans ]after, date]. */
function paymentsBetween(loan: Loan, after: ISODate | null, date: ISODate): number {
  let k = 0;
  for (let j = 0; j < loan.durationMonths; j++) {
    const d = addMonths(loan.startDate, j);
    if (d > date) break;
    if (!after || d > after) k++;
  }
  return k;
}

/** Capital restant dû à la date, quote-part appliquée. Le prêt existe à partir d'un mois avant la 1re échéance. */
export function outstandingAt(loan: Loan, date: ISODate): number {
  if (date < addMonths(loan.startDate, -1)) return 0;
  const i = loan.annualRate / 12;
  const M = monthlyPayment(loan.principal, loan.annualRate, loan.durationMonths);
  const manual = lastEntryOnOrBefore(loan.manualOutstanding, date);
  const b = manual
    ? balanceAfter(manual.amount, i, M, paymentsBetween(loan, manual.date, date))
    : balanceAfter(loan.principal, i, M, paymentsBetween(loan, null, date));
  return Math.round(b * (loan.ownershipPct ?? 1) * 100) / 100;
}

export interface LoanSummary { monthly: number; monthlyWithInsurance: number; endDate: ISODate; totalInterest: number; totalCost: number; outstanding: number; paidPct: number }

export function loanSummary(loan: Loan, date: ISODate): LoanSummary {
  const monthly = Math.round(monthlyPayment(loan.principal, loan.annualRate, loan.durationMonths) * 100) / 100;
  const ins = loan.insuranceMonthly ?? 0;
  const totalInterest = Math.round((monthly * loan.durationMonths - loan.principal) * 100) / 100;
  const own = loan.ownershipPct ?? 1;
  const outstanding = outstandingAt(loan, date);
  return {
    monthly, monthlyWithInsurance: monthly + ins,
    endDate: addMonths(loan.startDate, loan.durationMonths - 1),
    totalInterest, totalCost: totalInterest + ins * loan.durationMonths,
    outstanding, paidPct: 1 - outstanding / (loan.principal * own),
  };
}

export interface AmortizationRow { date: ISODate; interest: number; principal: number; balance: number }

export function amortizationSchedule(loan: Loan): AmortizationRow[] {
  const i = loan.annualRate / 12;
  const M = monthlyPayment(loan.principal, loan.annualRate, loan.durationMonths);
  let b = loan.principal;
  const rows: AmortizationRow[] = [];
  for (let k = 0; k < loan.durationMonths; k++) {
    const interest = b * i;
    const princ = Math.min(b, M - interest);
    b = Math.max(0, b - princ);
    rows.push({ date: addMonths(loan.startDate, k), interest, principal: princ, balance: Math.round(b * 100) / 100 });
  }
  return rows;
}

// file: src/core/assets.ts
import { lastEntryOnOrBefore, yearsBetween, daysBetween, type ISODate } from './dates';
import type { PhysicalAsset, RealEstate } from './types';

export function physicalValueAt(a: PhysicalAsset, date: ISODate): number {
  if (date < a.purchaseDate) return 0;
  const manual = lastEntryOnOrBefore(a.valuations, date);
  const baseDate = manual?.date ?? a.purchaseDate;
  const baseValue = manual?.value ?? a.purchasePrice;
  const y = yearsBetween(baseDate, date);
  const d = a.depreciation;
  let v: number;
  if (d.model === 'DEGRESSIVE') v = baseValue * Math.pow(1 - d.annualRate, y);
  else if (d.model === 'LINEAR') {
    const residual = a.purchasePrice * d.residualPct;
    const remaining = d.years - yearsBetween(a.purchaseDate, baseDate);
    v = remaining <= 0 ? Math.min(baseValue, residual) : baseValue - (baseValue - residual) * Math.min(1, y / remaining);
    v = Math.max(v, Math.min(baseValue, residual));
  } else v = baseValue;
  return Math.round(v * 100) / 100;
}

export function realEstateValueAt(r: RealEstate, date: ISODate): number {
  if (date < r.purchaseDate) return 0;
  const pts = [{ date: r.purchaseDate, value: r.purchasePrice }, ...[...r.valuations].sort((a, b) => a.date.localeCompare(b.date))];
  let v = pts[pts.length - 1].value;
  for (let k = 0; k < pts.length - 1; k++) {
    const a = pts[k], b = pts[k + 1];
    if (date >= a.date && date < b.date) {
      v = a.value + ((b.value - a.value) * daysBetween(a.date, date)) / daysBetween(a.date, b.date);
      break;
    }
  }
  return Math.round(v * r.ownershipPct * 100) / 100;
}

/** Rendements annuels sur le coût total (prix + frais + travaux), à 100 % du bien. */
export function realEstateYields(r: RealEstate): { gross: number; net: number; cost: number } {
  const cost = r.purchasePrice + (r.notaryFees ?? 0) + (r.worksCost ?? 0);
  const rent = (r.monthlyRent ?? 0) * 12;
  const net = rent - (r.monthlyCharges ?? 0) * 12 - (r.annualPropertyTax ?? 0);
  return { gross: cost ? rent / cost : 0, net: cost ? net / cost : 0, cost };
}

// file: src/core/valuation.ts
import type { ISODate } from './dates';
import { runLedger } from './positions';
import type { PriceBook } from './prices';
import { physicalValueAt, realEstateValueAt } from './assets';
import { outstandingAt } from './loans';
import { balanceAt } from './savings';
import type { AssetClass, GroupKey, NetWorthBreakdown, PositionView, Transaction, Vault } from './types';

export interface ValuationContext { vault: Vault; book: PriceBook; transactions: Transaction[] }

const add = <K extends string>(r: Partial<Record<K, number>>, k: K, v: number) => { if (v) r[k] = (r[k] ?? 0) + v; };

export function valueAt(ctx: ValuationContext, date: ISODate): NetWorthBreakdown {
  const { vault, book } = ctx;
  const accounts = new Map(vault.accounts.map((a) => [a.id, a]));
  const instruments = new Map(vault.instruments.map((i) => [i.id, i]));
  const ledger = runLedger(ctx.transactions, accounts, instruments, book, date);

  const byClass: Partial<Record<AssetClass, number>> = {};
  const byGroup: Partial<Record<GroupKey, number>> = {};
  const byAccount: Record<string, number> = {};
  const debtsByLoan: Record<string, number> = {};
  const positions: PositionView[] = [];

  for (const p of ledger.positions) {
    if (p.quantity <= 0) continue;
    const i = instruments.get(p.instrumentId)!;
    const a = accounts.get(p.accountId);
    const fx = book.fxAt(i.currency, date);
    const pru = p.costBasis / p.quantity;
    const pp = book.priceAt(i, date);
    const price = pp?.price ?? pru / fx;
    const value = p.quantity * price * fx;
    positions.push({
      accountId: p.accountId, instrumentId: i.id, name: i.name, assetClass: i.assetClass,
      quantity: p.quantity, pru, price, currency: i.currency,
      priceAsOf: pp?.asOf ?? date, priceSource: pp?.source ?? 'cost',
      value, costBasis: p.costBasis, unrealized: value - p.costBasis,
      unrealizedPct: p.costBasis ? (value - p.costBasis) / p.costBasis : 0,
      realized: p.realized, dividends: p.dividends, weight: 0,
      changePct1d: date >= book.today ? book.changePct1d(i) : undefined,
    });
    add(byClass, i.assetClass, value);
    add(byAccount, p.accountId, value);
    if (a) add(byGroup, a.type, value);
  }

  const cashByAccount: Record<string, number> = {};
  for (const a of vault.accounts) {
    const cash = ledger.cash[a.id] ?? 0;
    const bal = balanceAt(a, date);
    if (cash) { cashByAccount[a.id] = cash; add(byClass, 'CASH', cash); }
    add(byClass, a.type === 'ASSURANCE_VIE' || a.type === 'PER' ? 'OBLIGATIONS' : 'CASH', bal);
    add(byAccount, a.id, cash + bal);
    add(byGroup, a.type, cash + bal);
  }

  for (const r of vault.realEstate) {
    const v = realEstateValueAt(r, date);
    add(byClass, 'IMMOBILIER', v); add(byGroup, 'IMMOBILIER', v); add(byAccount, r.id, v);
  }
  for (const p of vault.physicalAssets) {
    const v = physicalValueAt(p, date);
    add(byClass, 'BIENS', v); add(byGroup, 'BIENS', v); add(byAccount, p.id, v);
  }
  let debts = 0;
  for (const l of vault.loans) {
    const o = outstandingAt(l, date);
    if (o) { debtsByLoan[l.id] = o; debts += o; }
  }

  const gross = Object.values(byClass).reduce((s, v) => s + (v ?? 0), 0);
  for (const p of positions) p.weight = gross ? p.value / gross : 0;
  positions.sort((a, b) => b.value - a.value);
  const invested = Object.entries(ledger.netDeposits).reduce((s, [, v]) => s + v, 0);

  return { date, gross, debts, net: gross - debts, byClass, byGroup, byAccount, cashByAccount, debtsByLoan, positions, invested };
}

// file: src/core/performance.ts
import { addMonths, daysBetween, type ISODate } from './dates';
import { balanceAt } from './savings';
import type { ValuationContext } from './valuation';
import type { Id } from './types';

export interface Flow { date: ISODate; amount: number }

export function xirr(flows: Flow[]): number | null {
  if (flows.length < 2 || !flows.some((f) => f.amount < 0) || !flows.some((f) => f.amount > 0)) return null;
  const t0 = flows.reduce((m, f) => (f.date < m ? f.date : m), flows[0].date);
  const ts = flows.map((f) => daysBetween(t0, f.date) / 365);
  const npv = (r: number) => flows.reduce((s, f, k) => s + f.amount / Math.pow(1 + r, ts[k]), 0);
  const dnpv = (r: number) => flows.reduce((s, f, k) => s - (ts[k] * f.amount) / Math.pow(1 + r, ts[k] + 1), 0);
  let r = 0.1;
  for (let k = 0; k < 100; k++) {
    const v = npv(r), d = dnpv(r);
    if (!isFinite(v) || !isFinite(d) || d === 0) break;
    const next = r - v / d;
    if (next <= -0.9999 || !isFinite(next)) break;
    if (Math.abs(next - r) < 1e-7) return next;
    r = next;
  }
  let lo = -0.9999, hi = 10;
  let flo = npv(lo);
  if (flo * npv(hi) > 0) return null;
  for (let k = 0; k < 200; k++) {
    const mid = (lo + hi) / 2, fm = npv(mid);
    if (Math.abs(fm) < 1e-7 || hi - lo < 1e-9) return mid;
    if (flo * fm < 0) hi = mid; else { lo = mid; flo = fm; }
  }
  return (lo + hi) / 2;
}

/** Argent neuf entré dans le patrimoine sur ]from, to] (EUR). */
export function externalFlows(ctx: ValuationContext, from: ISODate, to: ISODate): number {
  const { vault, book } = ctx;
  const ccy = new Map(vault.accounts.map((a) => [a.id, a.currency]));
  let f = 0;
  for (const t of ctx.transactions) {
    if (t.date <= from || t.date > to) continue;
    const fx = book.fxAt(ccy.get(t.accountId) ?? 'EUR', t.date);
    if (t.type === 'DEPOSIT') f += (t.amount ?? 0) * fx;
    else if (t.type === 'WITHDRAWAL') f -= (t.amount ?? 0) * fx;
  }
  for (const a of vault.accounts) if (a.balanceHistory?.length) f += balanceAt(a, to) - balanceAt(a, from);
  for (const r of vault.realEstate) if (r.purchaseDate > from && r.purchaseDate <= to) f += r.purchasePrice * r.ownershipPct;
  for (const p of vault.physicalAssets) if (p.purchaseDate > from && p.purchaseDate <= to) f += p.purchasePrice;
  for (const l of vault.loans) {
    const d = addMonths(l.startDate, -1);
    if (d > from && d <= to) f -= l.principal * (l.ownershipPct ?? 1);
  }
  return f;
}

export interface PeriodChange { change: number; flows: number; changeExFlows: number; pct: number | null }

/** Variation : brute, et hors apports (Dietz simplifié pour le %). */
export function periodChange(netFrom: number, netTo: number, flows: number): PeriodChange {
  const change = netTo - netFrom;
  const changeExFlows = change - flows;
  const denom = netFrom + flows / 2;
  return { change, flows, changeExFlows, pct: denom > 0 ? changeExFlows / denom : null };
}

/** XIRR des comptes donnés : versements (−), retraits (+), valeur finale (+). */
export function accountXirr(ctx: ValuationContext, accountIds: Id[], currentValue: number, date: ISODate): number | null {
  const ids = new Set(accountIds);
  const ccy = new Map(ctx.vault.accounts.map((a) => [a.id, a.currency]));
  const flows: Flow[] = [];
  for (const t of ctx.transactions) {
    if (!ids.has(t.accountId) || t.date > date) continue;
    const fx = ctx.book.fxAt(ccy.get(t.accountId) ?? 'EUR', t.date);
    if (t.type === 'DEPOSIT') flows.push({ date: t.date, amount: -(t.amount ?? 0) * fx });
    else if (t.type === 'WITHDRAWAL') flows.push({ date: t.date, amount: (t.amount ?? 0) * fx });
  }
  flows.push({ date, amount: currentValue });
  return xirr(flows);
}

import { addDays, addMonths, type ISODate } from './dates';
import { valueAt, type ValuationContext } from './valuation';
import type { NetWorthBreakdown, Snapshot, Vault } from './types';

export const toSnapshot = (b: NetWorthBreakdown): Snapshot => ({
  date: b.date, net: b.net, gross: b.gross, debts: b.debts, byClass: b.byClass, byAccount: b.byAccount, invested: b.invested,
});

export function earliestDate(v: Vault): ISODate | null {
  const ds: ISODate[] = [
    ...v.transactions.map((t) => t.date),
    ...v.dcaPlans.map((p) => p.startDate),
    ...v.realEstate.map((r) => r.purchaseDate),
    ...v.physicalAssets.map((p) => p.purchaseDate),
    ...v.loans.map((l) => addMonths(l.startDate, -1)),
    ...v.accounts.flatMap((a) => (a.balanceHistory ?? []).map((b) => b.date)),
  ];
  return ds.length ? ds.reduce((m, d) => (d < m ? d : m)) : null;
}

/** Quotidien sur les 365 derniers jours, hebdomadaire avant (aligné sur `to`). */
export function sampleDates(from: ISODate, to: ISODate): ISODate[] {
  const daily = addDays(to, -365);
  const out: ISODate[] = [];
  for (let d = to; d >= from; d = addDays(d, d > daily ? -1 : -7)) out.push(d);
  return out.reverse();
}

export function rebuildHistory(ctx: ValuationContext, from: ISODate, to: ISODate): Snapshot[] {
  return sampleDates(from, to).map((d) => toSnapshot(valueAt(ctx, d)));
}

import type { ISODate } from './dates';

// file: src/core/alerts.ts
import { addDays, daysBetween, lastEntryOnOrBefore } from './dates';
import { priceKey } from './prices';
import { ceilingFor, balanceAt } from './savings';
import type { Engine } from './engine';
import { ASSET_CLASS_LABELS, type AssetClass, type Id } from './types';

export type AlertLevel = 'info' | 'warn' | 'critical';
export type AlertFamily = 'MOVES' | 'ALLOCATION' | 'PRICE' | 'CONCENTRATION' | 'CEILING' | 'EMERGENCY' | 'DATA';
export const ALERT_FAMILY_LABELS: Record<AlertFamily, string> = {
  MOVES: 'Fortes variations', ALLOCATION: 'Allocation', PRICE: 'Seuils de prix', CONCENTRATION: 'Concentration',
  CEILING: 'Plafonds', EMERGENCY: 'Épargne de précaution', DATA: 'Données',
};

export interface Alert {
  id: string; family: AlertFamily; level: AlertLevel; title: string; detail: string;
  ref?: { kind: 'instrument' | 'account' | 'plan' | 'loan'; id: Id };
}

export interface Mover { instrumentId: Id; name: string; pct: number; value: number }

const pctFmt = (x: number) => `${x >= 0 ? '+' : ''}${(x * 100).toFixed(1).replace('.', ',')} %`;
const eurFmt = (x: number) => `${Math.round(x).toLocaleString('fr-FR')} €`;

/** Variation de cours (pas de valeur) des instruments détenus sur `days` jours. */
export function movers(e: Engine, days: number): Mover[] {
  const agg = new Map<Id, Mover>();
  const from = addDays(e.today, -days);
  for (const p of e.current.positions) {
    const i = e.vault.instruments.find((x) => x.id === p.instrumentId)!;
    const then = e.book.priceAt(i, from);
    const m = agg.get(i.id) ?? { instrumentId: i.id, name: i.name, pct: NaN, value: 0 };
    m.value += p.value;
    if (then && then.price > 0) m.pct = p.price / then.price - 1;
    agg.set(i.id, m);
  }
  return [...agg.values()].filter((m) => isFinite(m.pct)).sort((a, b) => b.pct - a.pct);
}

export function computeAlerts(e: Engine): Alert[] {
  const { vault, current: cur, today } = e;
  const th = vault.settings.alertThresholds;
  const out: Alert[] = [];
  const instruments = new Map(vault.instruments.map((i) => [i.id, i]));

  // 1. Variations
  const m7 = new Map(movers(e, 7).map((m) => [m.instrumentId, m]));
  for (const m of movers(e, 30)) {
    const w = m7.get(m.instrumentId);
    if (w && Math.abs(w.pct) >= th.weeklyMovePct) {
      out.push({ id: `move:${m.instrumentId}`, family: 'MOVES', level: 'warn', title: `${m.name} ${pctFmt(w.pct)} sur 7 jours`, detail: `Position : ${eurFmt(m.value)}`, ref: { kind: 'instrument', id: m.instrumentId } });
    } else if (Math.abs(m.pct) >= th.monthlyMovePct) {
      out.push({ id: `move:${m.instrumentId}`, family: 'MOVES', level: 'info', title: `${m.name} ${pctFmt(m.pct)} sur 30 jours`, detail: `Position : ${eurFmt(m.value)}`, ref: { kind: 'instrument', id: m.instrumentId } });
    }
  }

  // 2. Allocation
  const target = vault.settings.targetAllocation as Partial<Record<AssetClass, number>>;
  if (cur.gross > 0 && Object.keys(target).length) {
    let worst: { c: AssetClass; d: number } | null = null;
    for (const [c, t] of Object.entries(target) as [AssetClass, number][]) {
      const actual = (cur.byClass[c] ?? 0) / cur.gross;
      const d = actual - t;
      if (Math.abs(d) >= th.allocationDriftPts) {
        out.push({ id: `drift:${c}`, family: 'ALLOCATION', level: 'warn', title: `${ASSET_CLASS_LABELS[c]} : ${(actual * 100).toFixed(0)} % (cible ${(t * 100).toFixed(0)} %)`, detail: `Écart de ${pctFmt(d).replace(' %', ' pts')}` });
      }
      if (!worst || d < worst.d) worst = { c, d };
    }
    if (worst && worst.d <= -th.allocationDriftPts) {
      out.push({ id: 'drift:suggestion', family: 'ALLOCATION', level: 'info', title: `Orienter les prochains versements vers : ${ASSET_CLASS_LABELS[worst.c]}`, detail: `Il manque environ ${eurFmt(-worst.d * cur.gross)} pour revenir à la cible.` });
    }
  }

  // 3. Seuils de prix
  for (const a of vault.priceAlerts) {
    if (!a.active) continue;
    const i = instruments.get(a.instrumentId);
    if (!i) continue;
    const ps = cur.positions.filter((p) => p.instrumentId === i.id);
    const price = e.book.priceAt(i, today)?.price;
    const cost = ps.reduce((s, p) => s + p.costBasis, 0);
    const gain = cost ? ps.reduce((s, p) => s + p.value, 0) / cost - 1 : undefined;
    const hit =
      (a.condition === 'PRICE_BELOW' && price !== undefined && price < a.threshold) ||
      (a.condition === 'PRICE_ABOVE' && price !== undefined && price > a.threshold) ||
      (a.condition === 'GAIN_PCT_ABOVE' && gain !== undefined && gain > a.threshold) ||
      (a.condition === 'LOSS_PCT_BELOW' && gain !== undefined && gain < a.threshold);
    if (hit) {
      const what = a.condition.startsWith('PRICE') ? `cours ${price?.toLocaleString('fr-FR')} ${i.currency}` : `PV latente ${pctFmt(gain!)}`;
      out.push({ id: `price:${a.id}`, family: 'PRICE', level: 'warn', title: `${i.name} : seuil atteint`, detail: `${what}${a.note ? ` — ${a.note}` : ''}`, ref: { kind: 'instrument', id: i.id } });
    }
  }

  // 4. Concentration (par instrument, tous comptes)
  const byInst = new Map<Id, { name: string; value: number }>();
  for (const p of cur.positions) {
    const x = byInst.get(p.instrumentId) ?? { name: p.name, value: 0 };
    x.value += p.value; byInst.set(p.instrumentId, x);
  }
  for (const [id, x] of byInst) {
    const w = cur.gross ? x.value / cur.gross : 0;
    if (w > th.concentrationPct) out.push({ id: `concentration:${id}`, family: 'CONCENTRATION', level: 'warn', title: `${x.name} pèse ${(w * 100).toFixed(1).replace('.', ',')} % du patrimoine`, detail: `Seuil : ${(th.concentrationPct * 100).toFixed(0)} %`, ref: { kind: 'instrument', id } });
  }

  // 5. Plafonds
  const deposits = (accId: Id) => e.transactions.filter((t) => t.accountId === accId && t.type === 'DEPOSIT').reduce((s, t) => s + (t.amount ?? 0), 0);
  const peaDeposits = vault.accounts.filter((a) => a.type === 'PEA').reduce((s, a) => s + deposits(a.id), 0);
  for (const a of vault.accounts) {
    const ceiling = ceilingFor(a);
    if (!ceiling || a.archived) continue;
    const used = a.type === 'PEA' ? deposits(a.id) : a.type === 'PEA_PME' ? deposits(a.id) + peaDeposits : balanceAt(a, today);
    const r = used / ceiling;
    if (r >= th.ceilingWarnPct) {
      out.push({ id: `ceiling:${a.id}`, family: 'CEILING', level: r >= 1 ? 'critical' : 'warn', title: `${a.name} à ${(r * 100).toFixed(0)} % du plafond`, detail: `${eurFmt(used)} / ${eurFmt(ceiling)}`, ref: { kind: 'account', id: a.id } });
    }
  }

  // 6. Épargne de précaution
  if (th.emergencyFundMin && vault.settings.emergencyFundAccountIds.length) {
    const s = vault.settings.emergencyFundAccountIds.reduce((t, id) => t + (cur.byAccount[id] ?? 0), 0);
    if (s < th.emergencyFundMin) out.push({ id: 'emergency', family: 'EMERGENCY', level: 'warn', title: `Épargne de précaution sous le minimum`, detail: `${eurFmt(s)} / ${eurFmt(th.emergencyFundMin)}` });
  }

  // 7. Données
  const stale = cur.positions.filter((p) => {
    const i = instruments.get(p.instrumentId)!;
    return priceKey(i) && (p.priceSource === 'transaction' || p.priceSource === 'cost' || daysBetween(p.priceAsOf.slice(0, 10), today) > 4);
  });
  if (stale.length) out.push({ id: 'data:stale', family: 'DATA', level: 'info', title: `${stale.length} cours non à jour`, detail: stale.map((p) => p.name).join(', ') });
  if (e.pendingDca.length) out.push({ id: 'data:pending-dca', family: 'DATA', level: 'warn', title: `${e.pendingDca.length} échéance(s) DCA sans cours`, detail: e.pendingDca.map((p) => `${vault.dcaPlans.find((x) => x.id === p.planId)?.name} (${p.date})`).join(', ') });
  if (e.prices.errors.length) out.push({ id: 'data:errors', family: 'DATA', level: 'warn', title: `${e.prices.errors.length} erreur(s) de récupération de cours`, detail: e.prices.errors.map((x) => `${x.symbol} : ${x.message}`).join(' · ') });
  for (const l of vault.loans) {
    const last = lastEntryOnOrBefore(l.manualOutstanding, today)?.date ?? l.startDate;
    if (daysBetween(last, today) > 365 && (cur.debtsByLoan[l.id] ?? 0) > 0) {
      out.push({ id: `data:loan:${l.id}`, family: 'DATA', level: 'info', title: `Vérifier le capital restant dû de « ${l.name} »`, detail: 'Calculé depuis plus d\'un an sans valeur saisie (remboursement anticipé ?).', ref: { kind: 'loan', id: l.id } });
    }
  }

  const rank = { critical: 0, warn: 1, info: 2 };
  return out.sort((a, b) => rank[a.level] - rank[b.level]);
}

// file: src/core/report.ts
import { addDays, addMonths, type ISODate } from './dates';
import { computeAlerts, movers, type Alert, type Mover } from './alerts';
import { scheduleDates } from './dca';
import type { Engine } from './engine';
import { externalFlows, periodChange, type PeriodChange } from './performance';
import { valueAt } from './valuation';
import { ASSET_CLASSES, type AssetClass, type PendingDca } from './types';

export interface DcaLine { planName: string; date: ISODate; amount: number }
export interface WeeklyReport {
  date: ISODate; net: number; gross: number; debts: number;
  changes: (PeriodChange & { label: string })[];
  weekly: { date: ISODate; net: number }[];
  allocation: { cls: AssetClass; actual: number; target?: number }[];
  top: Mover[]; flop: Mover[];
  alerts: Alert[];
  dcaThisWeek: DcaLine[]; dcaNextWeek: DcaLine[]; pending: PendingDca[];
  pricesUpdatedAt: string; priceErrors: number;
}

export function buildWeeklyReport(e: Engine): WeeklyReport {
  const { today, current: cur, ctx, vault } = e;
  const period = (label: string, from: ISODate) => ({ label, ...periodChange(valueAt(ctx, from).net, cur.net, externalFlows(ctx, from, today)) });
  const weekly = Array.from({ length: 12 }, (_, k) => addDays(today, -7 * (11 - k))).map((d) => ({ date: d, net: d === today ? cur.net : valueAt(ctx, d).net }));
  const target = vault.settings.targetAllocation as Partial<Record<AssetClass, number>>;
  const allocation = ASSET_CLASSES
    .filter((c) => (cur.byClass[c] ?? 0) > 0 || target[c] !== undefined)
    .map((c) => ({ cls: c, actual: cur.gross ? (cur.byClass[c] ?? 0) / cur.gross : 0, target: target[c] }));
  const mv = movers(e, 7);
  const dca = (from: ISODate, to: ISODate): DcaLine[] => vault.dcaPlans.flatMap((p) =>
    scheduleDates(p, from, to).map((date) => ({ planName: p.name, date, amount: p.amount })))
    .sort((a, b) => a.date.localeCompare(b.date));
  return {
    date: today, net: cur.net, gross: cur.gross, debts: cur.debts,
    changes: [period('Semaine', addDays(today, -7)), period('Mois', addMonths(today, -1)), period('Depuis le 1er janvier', `${Number(today.slice(0, 4)) - 1}-12-31`)],
    weekly, allocation,
    top: mv.filter((m) => m.pct > 0).slice(0, 3),
    flop: mv.filter((m) => m.pct < 0).reverse().slice(0, 3),
    alerts: computeAlerts(e).filter((a) => a.level !== 'info'),
    dcaThisWeek: dca(addDays(today, -6), today),
    dcaNextWeek: dca(addDays(today, 1), addDays(today, 7)),
    pending: e.pendingDca,
    pricesUpdatedAt: e.prices.updatedAt, priceErrors: e.prices.errors.length,
  };
}
